cmake_minimum_required(VERSION 3.20)
project(DaisyLaunch VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus Concurrent)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6KIO REQUIRED)
set(DAISY_KDE_LIBRARIES Qt6::Core Qt6::Gui Qt6::DBus Qt6::Concurrent KF6::Service KF6::CoreAddons KF6::KIOGui KF6::ConfigCore)
option(DAISY_LAUNCH_LEGACY "Build the resident Qt/KDE launcher for A/B comparison" OFF)
if(DAISY_LAUNCH_LEGACY)
  add_executable(daisy-launch main.cpp)
  target_link_libraries(daisy-launch PRIVATE ${DAISY_KDE_LIBRARIES})
  install(TARGETS daisy-launch RUNTIME DESTINATION bin)
else()
  add_executable(daisy-launch-backend main.cpp)
  target_compile_definitions(daisy-launch-backend PRIVATE DAISY_LAUNCH_BACKEND=1)
  target_link_libraries(daisy-launch-backend PRIVATE ${DAISY_KDE_LIBRARIES})
  find_program(CARGO cargo REQUIRED)
  file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/rust/src/*.rs")
  set(RUST_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml")
  set(RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/cargo")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/daisy-launch"
    COMMAND ${CARGO} build --manifest-path "${RUST_MANIFEST}" --target-dir "${RUST_TARGET}" --release --locked
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RUST_TARGET}/release/daisy-launch" "${CMAKE_CURRENT_BINARY_DIR}/daisy-launch"
    DEPENDS ${RUST_SOURCES} "${RUST_MANIFEST}" "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.lock"
    VERBATIM)
  add_custom_target(daisy-launch-rust ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/daisy-launch")
  install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/daisy-launch" DESTINATION bin)
  install(TARGETS daisy-launch-backend RUNTIME DESTINATION bin)
endif()
