cmake_minimum_required(VERSION 3.22)
project(daisy_busy_cursor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS DBus Widgets)
add_library(daisy_busy_cursor MODULE main.cpp)
target_link_libraries(daisy_busy_cursor PRIVATE KWin::kwin Qt6::DBus)
set_target_properties(daisy_busy_cursor PROPERTIES PREFIX "" OUTPUT_NAME "daisyos-busy-cursor-v1" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS daisy_busy_cursor LIBRARY DESTINATION lib/qt6/plugins/kwin/effects/plugins)
