cmake_minimum_required(VERSION 3.20)
project(daisy_package_helper C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ALPM REQUIRED IMPORTED_TARGET libalpm)
find_package(OpenSSL REQUIRED)
add_executable(daisy-package-helper main.c)
target_compile_definitions(daisy-package-helper PRIVATE _GNU_SOURCE)
target_compile_options(daisy-package-helper PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_libraries(daisy-package-helper PRIVATE PkgConfig::ALPM OpenSSL::Crypto)
install(TARGETS daisy-package-helper RUNTIME DESTINATION lib/daisyos)
install(FILES ../../../packaging/applications/essential-applications.conf DESTINATION share/daisyos)
