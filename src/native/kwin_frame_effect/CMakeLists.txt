cmake_minimum_required(VERSION 3.25)
project(daisy_frame_effect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(KWin 6.8 REQUIRED)
# Private compositor APIs: build against the running KWin, never ship a cached ABI.
if(NOT KWin_VERSION VERSION_LESS 6.9)
    message(FATAL_ERROR "Review the frame effect against the KWin 6.9 API before building")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets CorePrivate)
find_package(KF6Config REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(daisyframe MODULE frameeffect.cpp resources.qrc)
target_link_libraries(daisyframe PRIVATE KWin::kwin Qt6::CorePrivate KF6::ConfigCore KF6::CoreAddons)
set_target_properties(daisyframe PROPERTIES PREFIX "")
install(TARGETS daisyframe LIBRARY DESTINATION lib/qt6/plugins/kwin/effects/plugins)
