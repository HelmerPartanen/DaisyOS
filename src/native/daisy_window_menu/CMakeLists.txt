cmake_minimum_required(VERSION 3.22)
project(daisy_window_menu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS DBus Widgets Svg)
find_package(KF6Config REQUIRED)
add_library(daisy_window_menu MODULE main.cpp daisymenustyle.cpp daisymenustyle.h)
target_link_libraries(daisy_window_menu PRIVATE KWin::kwin Qt6::DBus Qt6::Widgets Qt6::Svg KF6::ConfigCore)
set_target_properties(daisy_window_menu PROPERTIES PREFIX "" OUTPUT_NAME "daisyos-window-menus-v1" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS daisy_window_menu LIBRARY DESTINATION lib/qt6/plugins/kwin/effects/plugins)
add_executable(daisy-menu-preview preview.cpp daisymenustyle.cpp daisymenustyle.h)
target_link_libraries(daisy-menu-preview PRIVATE Qt6::Widgets Qt6::Svg KF6::ConfigCore)
