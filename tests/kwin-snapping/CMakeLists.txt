cmake_minimum_required(VERSION 3.25)
project(daisy_snap_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin 6.8 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets CorePrivate)
find_package(KF6CoreAddons REQUIRED)
add_library(daisysnaptest MODULE probe.cpp)
set_target_properties(daisysnaptest PROPERTIES PREFIX "")
target_link_libraries(daisysnaptest PRIVATE KWin::kwin Qt6::CorePrivate KF6::CoreAddons)
